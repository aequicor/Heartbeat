  it('binds a Tower worker to its spawning turn and revokes on cancellation', async () => {
    const origins = new HeartbeatOrigins(async (turn) => turn === 0 ? 'turn-token' : undefined);
    installHeartbeatOrigins('session-spawn-test', origins);
    runAgent.mockImplementation(async (agent) => ({
      agentId: agent.agentId,
      turn: { id: 42 },
      completion: completion.promise,
    }) as unknown as AgentRunHandle);
    try {
      const result = await execute(WORKER_ARGS);
      expect(result.isError).toBeUndefined();
      expect(await origins.resolve('agent-7', 42)).toBe('turn-token');
      const task = registerTask.mock.calls[0]?.[0] as SubagentTask;
      const stop = new AbortController();
      void task.start({ signal: stop.signal, write: () => {}, settle: async () => {} } as never);
      stop.abort();
      expect(await origins.resolve('agent-7', 42)).toBeUndefined();
    } finally {
      removeHeartbeatOrigins('session-spawn-test', origins);
      completion.resolve({ summary: 'done' });
    }
  });
